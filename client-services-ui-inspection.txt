"use client";

import { ArrowLeft, ArrowRight, Search, ShoppingCart } from "lucide-react";
import { useRouter } from "next/navigation";
import { useMemo, useState } from "react";
import {
  filterChooserServices,
  servicesInFolder,
  type ChooserService,
  type ServiceChooserCatalog,
} from "../../../domain/client-portal/service-chooser";
import { useServiceChooser } from "../services/service-chooser-context";
import { HeaderPanel, HeaderPanelAnchor, HeaderPanelTrigger } from "./header-panel";
import { useHeaderPanels } from "./header-panels";

/**
 * The cart entry in the header, and the service chooser it opens.
 *
 * Previously a centered modal opened from the rail. It is now an anchored
 * header panel on the same footing as notifications and quick actions, which
 * is what lets the manager guarantee only one of them is ever open. The rail's
 * Services entry went back to being a plain link to the Store page, so there is
 * one cart-oriented entry point rather than two with different behaviour.
 *
 * Categories and services are the real, RLS-scoped catalog the layout already
 * loads. Choosing a service hands off to the existing detail route.
 */

function ServiceRow({
  service,
  onChoose,
}: {
  service: ChooserService;
  onChoose: (service: ChooserService) => void;
}) {
  return (
    <li>
      <button type="button" className="chooser-row" onClick={() => onChoose(service)}>
        <span className="chooser-row-main">
          <strong>{service.name}</strong>
          <span>
            {service.folderName ? `${service.folderName} ¬∑ ` : ""}
            {service.recurringLabel ?? service.priceLabel}
          </span>
        </span>
        <ArrowRight aria-hidden={true} />
      </button>
    </li>
  );
}

function ChooserBody({
  catalog,
  onChoose,
}: {
  catalog: ServiceChooserCatalog;
  onChoose: (service: ChooserService) => void;
}) {
  const [query, setQuery] = useState("");
  const [folderId, setFolderId] = useState<string | null>(null);

  const searching = query.trim().length > 0;

  const results = useMemo(() => {
    if (searching) return filterChooserServices(catalog.services, query);
    if (folderId) return servicesInFolder(catalog.services, folderId);
    return [];
  }, [catalog.services, searching, query, folderId]);

  const activeFolder = folderId
    ? catalog.folders.find((folder) => folder.id === folderId) ?? null
    : null;

  return (
    <div className="chooser">
      <header className="portal-header-panel-head">
        <div>
          <strong>Explore Wolf Forge Services</strong>
          <span>Choose what you want to invest in.</span>
        </div>
      </header>

      <div className="chooser-search">
        <Search aria-hidden={true} />
        <input
          type="search"
          value={query}
          placeholder="Search services..."
          aria-label="Search services"
          onChange={(event) => setQuery(event.target.value)}
        />
      </div>

      {searching ? (
        <div className="chooser-panel" key="search">
          {results.length === 0 ? (
            <p className="chooser-empty">
              No services match ‚Äú{query.trim()}‚Äù. Try a shorter word.
            </p>
          ) : (
            <>
              <p className="chooser-caption" role="status">
                {results.length} {results.length === 1 ? "service" : "services"} found
              </p>
              <ul className="chooser-list">
                {results.map((service) => (
                  <ServiceRow key={service.id} service={service} onChoose={onChoose} />
                ))}
              </ul>
            </>
          )}
        </div>
      ) : activeFolder ? (
        <div className="chooser-panel" key={activeFolder.id}>
          <button type="button" className="chooser-back" onClick={() => setFolderId(null)}>
            <ArrowLeft aria-hidden={true} />
            All categories
          </button>

          <p className="chooser-caption">{activeFolder.name}</p>

          <ul className="chooser-list">
            {results.map((service) => (
              <ServiceRow key={service.id} service={service} onChoose={onChoose} />
            ))}
          </ul>
        </div>
      ) : (
        <div className="chooser-panel" key="categories">
          <p className="chooser-caption">Categories</p>

          {catalog.folders.length === 0 ? (
            <p className="chooser-empty">No services are available to your account yet.</p>
          ) : (
            <ul className="chooser-grid">
              {catalog.folders.map((folder) => (
                <li key={folder.id}>
                  <button
                    type="button"
                    className="chooser-category"
                    onClick={() => setFolderId(folder.id)}
                  >
                    <span className="chooser-category-mark" aria-hidden="true">
                      <ShoppingCart />
                    </span>
                    <span className="chooser-category-copy">
                      <strong>{folder.name}</strong>
                      <span>
                        {folder.serviceCount}{" "}
                        {folder.serviceCount === 1 ? "service" : "services"}
                      </span>
                    </span>
                    <ArrowRight aria-hidden={true} />
                  </button>
                </li>
              ))}
            </ul>
          )}
        </div>
      )}
    </div>
  );
}

export function ClientServicesPanel() {
  const chooser = useServiceChooser();
  const panels = useHeaderPanels();
  const router = useRouter();

  if (!chooser?.available) return null;

  const { catalog } = chooser;
  const open = panels?.isOpen("services") ?? false;

  const choose = (service: ChooserService) => {
    panels?.close();
    router.push(service.href);
  };

  return (
    <HeaderPanelAnchor>
      <HeaderPanelTrigger panel="services" label="Explore services">
        <ShoppingCart aria-hidden={true} />
      </HeaderPanelTrigger>

      <HeaderPanel panel="services" label="Explore Wolf Forge Services" width="wide">
        {/*
         * Mounted only while open, so its query and category reset each time
         * the panel opens without an effect watching the open flag.
         */}
        {open ? <ChooserBody catalog={catalog} onChoose={choose} /> : null}
      </HeaderPanel>
    </HeaderPanelAnchor>
  );
}

"use client";

import {
  forwardRef,
  useCallback,
  useEffect,
  useRef,
  useState,
  type ReactNode,
} from "react";
import { useHeaderPanels, type HeaderPanelId } from "./header-panels";

/**
 * The shared header trigger and its anchored panel.
 *
 * Every floating surface in the client header renders through this pair, so
 * corner radius, shadow, spacing rhythm, entrance and exit animation, and close
 * behaviour are defined once. A panel that opted out of this would be the
 * "patchy" case this replaces.
 */

/** Must match the exit duration on .portal-header-panel in globals.css. */
const EXIT_MS = 150;

export const HeaderPanelTrigger = forwardRef<
  HTMLButtonElement,
  {
    panel: HeaderPanelId;
    label: string;
    children: ReactNode;
    className?: string;
    /** Rendered inside the trigger, e.g. the unread badge. */
    badge?: ReactNode;
  }
>(function HeaderPanelTrigger({ panel, label, children, className, badge }, ref) {
  const panels = useHeaderPanels();
  const open = panels?.isOpen(panel) ?? false;

  return (
    <button
      ref={ref}
      type="button"
      className={["portal-command-button", open ? "is-open" : "", className]
        .filter(Boolean)
        .join(" ")}
      aria-label={label}
      title={label}
      aria-haspopup="dialog"
      aria-expanded={open}
      onPointerEnter={(event) => {
        // Touch fires pointerenter immediately before click; the manager
        // already ignores hover on coarse pointers, and this keeps a stylus or
        // synthetic pointer from opening on contact.
        if (event.pointerType !== "mouse") return;
        panels?.hoverOpen(panel);
      }}
      onPointerLeave={() => panels?.hoverLeave()}
      onFocus={() => panels?.cancelClose()}
      onClick={() => panels?.toggle(panel)}
    >
      {children}
      {badge}
    </button>
  );
});

/**
 * The anchored panel itself.
 *
 * Kept mounted through its exit animation so closing actually plays. Hovering
 * the panel cancels the pending close, which is what lets the pointer travel
 * from the trigger down into the content.
 */
export function HeaderPanel({
  panel,
  label,
  children,
  width = "default",
}: {
  panel: HeaderPanelId;
  label: string;
  children: ReactNode;
  width?: "default" | "wide";
}) {
  const panels = useHeaderPanels();
  const open = panels?.isOpen(panel) ?? false;

  const [lastOpen, setLastOpen] = useState(open);
  const [exiting, setExiting] = useState(false);

  /* Adjust-state-on-prop-change: the closing frame has to render in the same
     pass the panel closed, which an effect would miss by one frame. */
  if (lastOpen !== open) {
    setLastOpen(open);
    setExiting(!open);
  }

  useEffect(() => {
    if (!exiting) return;
    const timer = setTimeout(() => setExiting(false), EXIT_MS);
    return () => clearTimeout(timer);
  }, [exiting]);

  const onPointerEnter = useCallback(() => panels?.cancelClose(), [panels]);
  const onPointerLeave = useCallback(() => panels?.hoverLeave(), [panels]);

  if (!open && !exiting) return null;

  return (
    <div
      className="portal-header-panel"
      data-phase={open ? "open" : "closing"}
      data-width={width}
      role="dialog"
      aria-label={label}
      onPointerEnter={onPointerEnter}
      onPointerLeave={onPointerLeave}
    >
      {children}
    </div>
  );
}

/**
 * Wraps one trigger and its panel.
 *
 * The marker attribute is what the manager's outside-click check looks for, so
 * a click anywhere inside a trigger or its panel is never "outside".
 */
export function HeaderPanelAnchor({
  children,
  className,
}: {
  children: ReactNode;
  className?: string;
}) {
  const ref = useRef<HTMLDivElement>(null);

  return (
    <div
      ref={ref}
      data-header-panel-root=""
      className={["portal-command-anchor", className].filter(Boolean).join(" ")}
    >
      {children}
    </div>
  );
}
