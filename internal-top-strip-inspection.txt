type RouteTransitionPhase =
  | "idle"
  | "vacuum"
  | "enter";

function SidebarLink({
  id,
  label,
  route,
  active,
  transitioning,
  onNavigate,
}: {
  id: string;
  label: string;
  route: string;
  active: boolean;
  transitioning: boolean;
  onNavigate: (
    route: string,
    event: MouseEvent<HTMLAnchorElement>,
  ) => void;
}) {
  const Icon =
    navigationIcons[id] ?? FileText;

  const className = [
    "sidebar-link",
    active ? "is-active" : "",
    transitioning
      ? "is-transition-target"
      : "",
  ]
    .filter(Boolean)
    .join(" ");

  return (
    <Link
      className={className}
      href={route}
      aria-current={
        active ? "page" : undefined
      }
      onClick={(event) =>
        onNavigate(route, event)
      }
    >
      <Icon aria-hidden={true} />
      <span>{label}</span>
    </Link>
  );
}

export function InternalShell({
  user,
  children,
}: {
  user: InternalUser;
  children: ReactNode;
}) {
  const pathname = usePathname();
  const router = useRouter();
  const navigationCategories = navigationForModules(user.modules);

  const mainRef =
    useRef<HTMLElement>(null);

  const pendingRouteRef =
    useRef<string | null>(null);

  const exitTimerRef =
    useRef<
      ReturnType<typeof setTimeout> | null
    >(null);

  const frameRefs =
    useRef<number[]>([]);

  const [
    transitionPhase,
    setTransitionPhase,
  ] =
    useState<RouteTransitionPhase>(
      "idle",
    );

  const [
    transitionTarget,
    setTransitionTarget,
  ] =
    useState<string | null>(null);


  useEffect(() => {
    return () => {
      if (exitTimerRef.current) {
        clearTimeout(
          exitTimerRef.current,
        );
      }

      frameRefs.current.forEach(
        (frame) =>
          window.cancelAnimationFrame(
            frame,
          ),
      );
    };
  }, []);


  useEffect(() => {
    const pending =
      pendingRouteRef.current;

    if (!pending) {
      return;
    }

    if (
      !routeIsCurrent(
        pending,
        pathname,
      )
    ) {
      return;
    }

    if (exitTimerRef.current) {
      clearTimeout(
        exitTimerRef.current,
      );

      exitTimerRef.current = null;
    }

    /*
     * The new route has committed.
     *
     * Keep it hidden for one paint,
     * then release it into the normal
     * workspace position.
     */
    setTransitionPhase("enter");

    frameRefs.current.forEach(
      (frame) =>
        window.cancelAnimationFrame(
          frame,
        ),
    );

    frameRefs.current = [];

    const firstFrame =
      window.requestAnimationFrame(
        () => {
          const secondFrame =
            window.requestAnimationFrame(
              () => {
                setTransitionPhase(
                  "idle",
                );

                setTransitionTarget(
                  null,
                );

                pendingRouteRef.current =
                  null;

                frameRefs.current = [];
              },
            );

          frameRefs.current = [
            secondFrame,
          ];
        },
      );

    frameRefs.current = [
      firstFrame,
    ];
  }, [pathname]);


  function handleSidebarNavigate(
    route: string,
    event: MouseEvent<HTMLAnchorElement>,
  ) {
    /*
     * Preserve normal browser behavior
     * for new tabs / modified clicks.
     */
    if (
      event.button !== 0 ||
      event.metaKey ||
      event.ctrlKey ||
      event.shiftKey ||
      event.altKey
    ) {
      return;
    }

    if (
      routeIsCurrent(
        route,
        pathname,
      )
    ) {
      event.preventDefault();
      return;
    }

    if (
      transitionPhase !== "idle"
    ) {
      event.preventDefault();
      return;
    }

    event.preventDefault();

    /*
     * Respect reduced-motion users and
     * navigate immediately.
     */
    if (
      window.matchMedia(
        "(prefers-reduced-motion: reduce)",
      ).matches
    ) {
      router.push(route);
      return;
    }

    const main =
      mainRef.current;

    if (!main) {
      router.push(route);
      return;
    }

    /*
     * Warm the destination while the
     * outgoing animation is playing.
     */
    router.prefetch(route);

    const destinationRect =
      event.currentTarget.getBoundingClientRect();

    const mainRect =
      main.getBoundingClientRect();

    const destinationCenterY =
      destinationRect.top +
      destinationRect.height / 2;

    /*
     * Make the page collapse toward the
     * vertical position of the sidebar
     * item that was clicked.
     */
    const rawOriginY =
      destinationCenterY -
      mainRect.top;

    const safeMinimum = 24;

    const safeMaximum =
      Math.max(
        safeMinimum,
        mainRect.height - 24,
      );

    const originY =
      Math.min(
        Math.max(
          rawOriginY,
          safeMinimum,
        ),
        safeMaximum,
      );

    main.style.setProperty(
      "--wf-route-origin-y",
      `${originY}px`,
    );

    pendingRouteRef.current =
      route;

    setTransitionTarget(route);

    setTransitionPhase(
      "vacuum",
    );

    const navigationFrame =
      window.requestAnimationFrame(
        () => {
          router.push(route);
        },
      );

    frameRefs.current.push(
      navigationFrame,
    );
  }


  const stageClassName = [
    "workspace-route-stage",

    transitionPhase === "vacuum"
      ? "is-vacuuming"
      : "",

    transitionPhase === "enter"
      ? "is-entering"
      : "",
  ]
    .filter(Boolean)
    .join(" ");


  return (
    <div className="internal-app">
      <a
        className="skip-link"
        href="#main-content"
      >
        Skip to content
      </a>

      <aside className="app-sidebar">
        <div
          className="sidebar-brand is-mark-only"
          aria-label="Wolf Forge"
        >
          <WolfForgeMark />
        </div>

        <nav
          className="sidebar-navigation"
          aria-label="Primary"
        >
          {navigationCategories.map(
            (category) => (
              <section
                className="sidebar-section"
                key={category.id}
                aria-labelledby={
                  `nav-${category.id}`
                }
              >
                <h2
                  id={`nav-${category.id}`}
                >
                  {category.label}
                </h2>

                <div className="sidebar-links">
                  {category.items.map(
                    (item) => (
                      <SidebarLink
                        key={item.id}
                        id={item.id}
                        label={item.label}
                        route={item.route}
                        active={routeIsCurrent(
                          item.route,
                          transitionTarget ?? pathname,
                        )}
                        transitioning={
                          transitionTarget ===
                          item.route
                        }
                        onNavigate={
                          handleSidebarNavigate
                        }
                      />
                    ),
                  )}
                </div>
              </section>
            ),
          )}
        </nav>

        <div className="sidebar-account">
          <AccountMenu
            user={user}
            sidebar
          />
        </div>
      </aside>

      <div
        id="main-content"
        className="workspace"
      >
        <WorkspaceCommandBar
          user={user}
        />

        <main
          ref={mainRef}
          className={
            stageClassName
          }
        >
          {children}
        </main>
      </div>
    </div>
  );
}

function formatActivityTime(
  value: string,
) {
  const date = new Date(value);

  return new Intl.DateTimeFormat(
    undefined,
    {
      month: "short",
      day: "numeric",
      hour: "numeric",
      minute: "2-digit",
    },
  ).format(date);
}

function NotificationCategoryIcon({
  category,
}: {
  category:
    WalletNotificationItem["category"];
}) {
  const Icon =
    category === "project"
      ? Folder
      : category === "client"
        ? UserPlus
        : category === "form"
          ? FileText
          : category === "coupon"
            ? Ticket
            : category === "ticket"
              ? MessageSquare
              : category === "payment"
                ? Receipt
                : category === "wallet"
                  ? WalletCards
                  : Bell;

  return (
    <Icon aria-hidden="true" />
  );
}
type NotificationPosition = {
  top: number;
  right: number;
};

export function WorkspaceCommandBar({
  user,
}: {
  user: InternalUser;
}) {
  const triggerRef =
    useRef<HTMLButtonElement>(null);

  const panelRef =
    useRef<HTMLElement>(null);

  const [
    mounted,
    setMounted,
  ] = useState(false);

  const [
    notifications,
    setNotifications,
  ] = useState<
    WalletNotificationItem[]
  >([]);

  const [
    notificationsOpen,
    setNotificationsOpen,
  ] = useState(false);

  const [
    notificationsLoading,
    setNotificationsLoading,
  ] = useState(false);

  const [
    notificationsError,
    setNotificationsError,
  ] = useState<string | null>(
    null,
  );

  const [
    notificationPosition,
    setNotificationPosition,
  ] =
    useState<NotificationPosition>({
      top: 76,
      right: 20,
    });

  const [
    notificationRinging,
    setNotificationRinging,
  ] = useState(false);

  const notificationSeenIdsRef =
    useRef<Set<string>>(
      new Set(),
    );

  const notificationFeedInitializedRef =
    useRef(false);

  const notificationFeedbackTimersRef =
    useRef<number[]>([]);

  const noteIncomingNotifications =
    useCallback(
      (
        items:
          WalletNotificationItem[],
      ) => {
        const nextIds =
          new Set(
            items.map(
              (item) =>
                item.id,
            ),
          );

        if (
          !notificationFeedInitializedRef.current
        ) {
          notificationSeenIdsRef.current =
            nextIds;

          notificationFeedInitializedRef.current =
            true;

          return;
        }

        const incoming =
          items.filter(
            (item) =>
              !item.readAt &&
              !notificationSeenIdsRef.current.has(
                item.id,
              ),
          );

        notificationSeenIdsRef.current =
          nextIds;

        incoming.forEach(
          (_, index) => {
            const startTimer =
              window.setTimeout(
                () => {
                  setNotificationRinging(
                    true,
                  );

                  playNotificationChime();

                  const stopTimer =
                    window.setTimeout(
                      () => {
                        setNotificationRinging(
                          false,
                        );
                      },
                      560,
                    );

                  notificationFeedbackTimersRef.current.push(
                    stopTimer,
                  );
                },
                index * 680,
              );

            notificationFeedbackTimersRef.current.push(
              startTimer,
            );
          },
        );
      },
      [],
    );

  const loadNotifications =
    useCallback(
      async (
        silent = false,
      ) => {
        if (!silent) {
          setNotificationsLoading(
            true,
          );
        }

        const result =
          await getWalletNotificationsAction();

        if (result.ok) {
          noteIncomingNotifications(
            result.notifications,
          );

          setNotifications(
            result.notifications,
          );

          setNotificationsError(
            null,
          );
        } else if (!silent) {
          setNotificationsError(
            result.error,
          );
        }

        if (!silent) {
          setNotificationsLoading(
            false,
          );
        }
      },
      [
        noteIncomingNotifications,
      ],
    );

  const updatePanelPosition =
    useCallback(() => {
      const trigger =
        triggerRef.current;

      if (!trigger) {
        return;
      }

      const rect =
        trigger.getBoundingClientRect();

      const right =
        Math.max(
          12,
          window.innerWidth -
            rect.right,
        );

      setNotificationPosition({
        top: rect.bottom + 10,
        right,
      });
    }, []);

  useEffect(() => {
    setMounted(true);

    void loadNotifications();
  }, [loadNotifications]);

  useEffect(() => {
    const unlock = () => {
      void unlockNotificationAudio();
    };

    window.addEventListener(
      "pointerdown",
      unlock,
      {
        once: true,
        passive: true,
      },
    );

    window.addEventListener(
      "keydown",
      unlock,
      {
        once: true,
      },
    );

    return () => {
      window.removeEventListener(
        "pointerdown",
        unlock,
      );

      window.removeEventListener(
        "keydown",
        unlock,
      );
    };
  }, []);

  useEffect(() => {
    const refresh = () => {
      if (!document.hidden) {
        void loadNotifications(
          true,
        );
      }
    };

    const interval =
      window.setInterval(
        refresh,
        4000,
      );

    window.addEventListener(
      "focus",
      refresh,
    );

    document.addEventListener(
      "visibilitychange",
      refresh,
    );

    return () => {
      window.clearInterval(
        interval,
      );

      window.removeEventListener(
        "focus",
        refresh,
      );

      document.removeEventListener(
        "visibilitychange",
        refresh,
      );
    };
  }, [loadNotifications]);

  useEffect(() => {
    return () => {
      notificationFeedbackTimersRef.current.forEach(
        (timer) => {
          window.clearTimeout(
            timer,
          );
        },
      );

      notificationFeedbackTimersRef.current =
        [];
    };
  }, []);

  useEffect(() => {
    if (!notificationsOpen) {
      return;
    }

    updatePanelPosition();

    const handleResize = () => {
      updatePanelPosition();
    };

    const handleScroll = () => {
      updatePanelPosition();
    };

    window.addEventListener(
      "resize",
      handleResize,
    );

    window.addEventListener(
      "scroll",
      handleScroll,
      true,
    );

    return () => {
      window.removeEventListener(
        "resize",
        handleResize,
      );

      window.removeEventListener(
        "scroll",
        handleScroll,
        true,
      );
    };
  }, [
    notificationsOpen,
    updatePanelPosition,
  ]);

  useEffect(() => {
    const closeForQuickAction = () => {
      setNotificationsOpen(false);
    };

    window.addEventListener(
      "wf:quick-action-open",
      closeForQuickAction,
    );

    return () => {
      window.removeEventListener(
        "wf:quick-action-open",
        closeForQuickAction,
      );
    };
  }, []);

  useEffect(() => {
    const handlePointerDown = (
      event: PointerEvent,
    ) => {
      if (!notificationsOpen) {
        return;
      }

      const target =
        event.target as Node;

      const trigger =
        triggerRef.current;

      const panel =
        panelRef.current;

      if (
        trigger?.contains(target) ||
        panel?.contains(target)
      ) {
        return;
      }

      setNotificationsOpen(false);
    };

    const handleKeyDown = (
      event: KeyboardEvent,
    ) => {
      if (
        event.key === "Escape"
      ) {
        setNotificationsOpen(
          false,
        );
      }
    };

    document.addEventListener(
      "pointerdown",
      handlePointerDown,
    );

    document.addEventListener(
      "keydown",
      handleKeyDown,
    );

    return () => {
      document.removeEventListener(
        "pointerdown",
        handlePointerDown,
      );

      document.removeEventListener(
        "keydown",
        handleKeyDown,
      );
    };
  }, [notificationsOpen]);

  const toggleNotifications =
    async () => {
      if (notificationsOpen) {
        setNotificationsOpen(false);
        return;
      }

      updatePanelPosition();

      setNotificationsOpen(true);

      await loadNotifications();
    };

  const unreadNotificationCount =
    notifications.reduce(
      (
        total,
        notification,
      ) =>
        total +
        (notification.readAt
          ? 0
          : 1),
      0,
    );

  const notificationBadgeLabel =
    unreadNotificationCount > 9
      ? "9+"
      : String(
          unreadNotificationCount,
        );
  const markAllAsRead = () => {
    if (
      unreadNotificationCount === 0
    ) {
      return;
    }

    const readAt =
      new Date().toISOString();

    setNotifications(
      (current) =>
        current.map(
          (notification) =>
            notification.readAt
              ? notification
              : {
                  ...notification,
                  readAt,
                },
        ),
    );

    void markAllNotificationsReadAction();
  };
  const notificationPanel = (
    <section
      ref={panelRef}
      className={
        notificationsOpen
          ? "workspace-notification-portal is-open"
          : "workspace-notification-portal"
      }
      style={{
        top:
          notificationPosition.top,
        right:
          notificationPosition.right,
      }}
      aria-hidden={
        !notificationsOpen
      }
      aria-label="Notifications"
    >
      <header className="workspace-notification-header">
        <div>
          <strong>
            Notifications
          </strong>

          <span>
            Recent activity
          </span>
        </div>

        <div className="workspace-notification-header-actions">
          {notificationsLoading ? (
            <span className="workspace-notification-refreshing">
              Refreshing...
            </span>
          ) : null}

          {unreadNotificationCount > 0 ? (
            <button
              type="button"
              className="workspace-notification-mark-all"
              onClick={markAllAsRead}
            >
              Mark all as read
            </button>
          ) : (
            <span className="workspace-notification-all-read">
              All read
            </span>
          )}
        </div>
      </header>

      <div className="workspace-notification-list">
        {notificationsError ? (
          <div className="workspace-notification-empty">
            {notificationsError}
          </div>
        ) : notifications.length ===
            0 &&
          !notificationsLoading ? (
          <div className="workspace-notification-empty">
            No activity yet.
          </div>
        ) : (
          notifications.map(
            (notification) => (
              <Link
                key={
                  notification.id
                }
                href={
                  notification.href
                }
                className="workspace-notification-item"
                onClick={() => {
                  setNotifications(
                    (current) =>
                      current.map(
                        (item) =>
                          item.id ===
                          notification.id
                            ? {
                                ...item,
                                readAt:
                                  item.readAt ??
                                  new Date().toISOString(),
                              }
                            : item,
                      ),
                  );

                  void markNotificationReadAction(
                    notification.id,
                  );

                  setNotificationsOpen(
                    false,
                  );
                }}
              >
                <span
                  className={
                    notification.kind ===
                    "credit"
                      ? "workspace-notification-kind is-credit"
                      : "workspace-notification-kind is-debit"
                  }
                >
                  <NotificationCategoryIcon
                    category={
                      notification.category
                    }
                  />
                </span>

                <span className="workspace-notification-copy">
                  <strong>
                    {
                      notification.title
                    }
                  </strong>

                  <span className="workspace-notification-body">
                    {
                      notification.body
                    }
                  </span>

                  {notification.reason ? (
                    <span className="workspace-notification-reason">
                      Reason:{" "}
                      {
                        notification.reason
                      }
                    </span>
                  ) : null}

                  <time
                    dateTime={
                      notification.createdAt
                    }
                  >
                    {formatActivityTime(
                      notification.createdAt,
                    )}
                  </time>
                </span>
              </Link>
            ),
          )
        )}
      </div>
    </section>
  );

  return (
    <>
      <div
        className="workspace-command-bar"
        aria-label="Workspace controls"
      >
        <GlobalSearch />
        <div className="workspace-command-shortcuts">
        <DashboardDateTrigger />
          <button
            className="workspace-command-button"
            type="button"
            aria-label="Gallery"
            title="Gallery"
          >
            <Images
              aria-hidden="true"
            />
          </button>

          <button
            ref={triggerRef}
            className={[
              "workspace-command-button",
              "workspace-notification-trigger",
              notificationsOpen
                ? "is-open"
                : "",
              notificationRinging
                ? "is-ringing"
                : "",
            ]
              .filter(Boolean)
              .join(" ")}
            type="button"
            aria-label="Notifications"
            title="Notifications"
            aria-expanded={
              notificationsOpen
            }
            aria-haspopup="dialog"
            onClick={
              toggleNotifications
            }
          >
            <Bell
              aria-hidden="true"
            />

            {notifications.some((notification) => !notification.readAt) ? (
              <span
                className="workspace-notification-badge"
                aria-label={`${notificationBadgeLabel} unread notifications`}
              >
                {notificationBadgeLabel}
              </span>
            ) : null}
          </button>

          <WorkspaceQuickAction
            user={user}
          />
        </div>

        <div className="workspace-command-profile">
          <AccountMenu
            user={user}
          />
        </div>
      </div>

      {mounted
        ? createPortal(
            notificationPanel,
            document.body,
          )
        : null}
    </>
  );
}
