type ServiceSort = "name" | "price" | "created";
type SortDirection = "asc" | "desc";

function flattenFolders(folders: ServiceFolderRow[]) {
  const result: Array<{ folder: ServiceFolderRow; depth: number }> = [];
  const visited = new Set<string>();

  function visit(parentId: string | null, depth: number) {
    for (const folder of folders.filter((candidate) => candidate.parent_id === parentId)) {
      if (visited.has(folder.id)) continue;
      visited.add(folder.id);
      result.push({ folder, depth });
      visit(folder.id, depth + 1);
    }
  }

  visit(null, 0);
  for (const folder of folders) {
    if (!visited.has(folder.id)) result.push({ folder, depth: 0 });
  }
  return result;
}

function formatMoney(service: ServiceRow) {
  return new Intl.NumberFormat("en-US", {
    style: "currency",
    currency: service.currency,
    minimumFractionDigits: service.price % 1 === 0 ? 0 : 2,
  }).format(service.price);
}

function formatCatalogDate(value: string) {
  return new Intl.DateTimeFormat("en", { day: "numeric", month: "short", year: "numeric" }).format(new Date(value));
}

function sortServices(services: ServiceRow[], sort: ServiceSort, direction: SortDirection) {
  const multiplier = direction === "asc" ? 1 : -1;
  return [...services].sort((left, right) => {
    if (sort === "price") return (left.price - right.price) * multiplier;
    if (sort === "created") return left.created_at.localeCompare(right.created_at) * multiplier;
    return left.name.localeCompare(right.name) * multiplier;
  });
}

function SortLink({
  label,
  column,
  currentSort,
  direction,
  folder,
  query,
}: {
  label: string;
  column: ServiceSort;
  currentSort: ServiceSort;
  direction: SortDirection;
  folder: string;
  query: string;
}) {
  const nextDirection = currentSort === column && direction === "asc" ? "desc" : "asc";
  const params = new URLSearchParams({ folder, sort: column, direction: nextDirection });
  if (query) params.set("q", query);
  return (
    <Link
      className="services-sort-link"
      href={`/services?${params}`}
    >
      <span>
        {label}
      </span>

      {currentSort === column ? (
        direction === "asc" ? (
          <ArrowUp
            aria-hidden="true"
          />
        ) : (
          <ArrowDown
            aria-hidden="true"
          />
        )
      ) : null}
    </Link>
  );
}

    q?: string;
    folder?: string;
    sort?: string;
    direction?: string;
    archived?: string;
    folderCreated?: string;
    folderArchived?: string;
  }>;
}) {
  const parameters = await searchParams;
  const q = parameters.q ?? "";
  const folderParameter = parameters.folder ?? "unfiled";
  const sort: ServiceSort = ["name", "price", "created"].includes(parameters.sort ?? "")
    ? parameters.sort as ServiceSort
    : "name";
  const direction: SortDirection = parameters.direction === "desc" ? "desc" : "asc";
  const context = await getInternalContext();

  try {
    const [services, folders] = await Promise.all([
      context.services.catalog.listCatalog(),
      context.services.catalog.listFolders(),
    ]);
    const selectedFolder = folders.find((folder) => folder.id === folderParameter) ?? null;
    const selectedFolderId = selectedFolder?.id ?? null;
    const needle = q.trim();

    const visibleServices =
      services.filter((service) => {
        /*
         * When there is an active search, search the ENTIRE
         * service catalog regardless of folder.
         *
         * Without a search, preserve normal folder behavior.
         */
        if (needle) {
          return (
            serviceNameSearchScore(
              service.name,
              needle,
            ) !== null
          );
        }

        return selectedFolderId
          ? service.folder_id === selectedFolderId
          : service.folder_id === null;
      });

    const sortedServices =
      needle
        ? [...visibleServices]
            .sort((left, right) =>
              compareServiceNameMatches(
                left.name,
                right.name,
                needle,
              ),
            )
            .slice(0, 200)
        : sortServices(
            visibleServices,
            sort,
            direction,
          ).slice(0, 200);
    const serviceCounts = new Map<string, number>();
    for (const service of services) {
      if (service.folder_id) serviceCounts.set(service.folder_id, (serviceCounts.get(service.folder_id) ?? 0) + 1);
    }
    const unfiledCount = services.filter((service) => service.folder_id === null).length;
    const folderQueryValue = selectedFolder?.id ?? "unfiled";

    return (
      <PageFrame
        title="Services"
        description="Configure the catalog that becomes Wolf Forge Project work."
        action={<ServicesHeaderActions />}
      >
        {parameters.archived ? <div className="success-banner">Service soft deleted.</div> : null}
        {parameters.folderCreated ? <div className="success-banner">Folder created.</div> : null}
        {parameters.folderArchived ? <div className="success-banner">Folder soft deleted. Its services are now Unfiled.</div> : null}

        <section className="services-folder-section" aria-labelledby="services-folders-heading">
          <div className="services-section-heading">
            <div><span>Catalog structure</span><h2 id="services-folders-heading">Folders</h2></div>
            <Link href="/services/folders/new"><Plus aria-hidden="true" />New Folder</Link>
          </div>
          <div className="services-folder-list">
            {flattenFolders(folders).map(({ folder, depth }) => (

                <Link className="services-folder-link" href={`/services?folder=${folder.id}`}>
                  <Folder aria-hidden="true" />

            <div className={!selectedFolder ? "services-folder-row is-active" : "services-folder-row"}>

export function RichTextDescription({
  initialValue,
  initialHtml,
}: {
  initialValue: string;
  initialHtml: string;
}) {
  const editorRef = useRef<HTMLDivElement>(null);
  const inputRef = useRef<HTMLInputElement>(null);

  function syncDescription() {
    const editor = editorRef.current;
    const input = inputRef.current;
    if (editor && input) input.value = editor.innerHTML;
  }

  function runCommand(command: string, value?: string) {
    const editor = editorRef.current;
    if (!editor) return;
    editor.focus();
    document.execCommand(command, false, value);
    syncDescription();
  }

  function createLink() {
    const href = window.prompt("Link URL", "https://");
    if (!href) return;

    try {
      const url = new URL(href, window.location.origin);
      if (!["http:", "https:", "mailto:", "tel:"].includes(url.protocol)) return;
      runCommand("createLink", href);
    } catch {
      // Ignore invalid URLs and leave the selected text unchanged.
    }
  }

  function pastePlainText(event: ClipboardEvent<HTMLDivElement>) {
    event.preventDefault();
    document.execCommand("insertText", false, event.clipboardData.getData("text/plain"));
    syncDescription();
  }

  return (
    <div className="service-rich-field">
      <div className="service-rich-toolbar" aria-label="Description formatting">
        <EditorButton label="Bold" icon={Bold} onClick={() => runCommand("bold")} />
        <EditorButton label="Italic" icon={Italic} onClick={() => runCommand("italic")} />
        <EditorButton label="Heading" icon={Heading2} onClick={() => runCommand("formatBlock", "h2")} />
        <span className="service-rich-toolbar-divider" aria-hidden="true" />

import { revalidatePath } from "next/cache";

export type CatalogFormState = {
  error: string | null;
};

function optionalText(formData: FormData, key: string) {
  const value = String(formData.get(key) ?? "").trim();
  return value || null;
}

function requiredNumber(formData: FormData, key: string) {
  const value = String(formData.get(key) ?? "").trim();
  return value === "" ? Number.NaN : Number(value);
}

function optionalNumber(formData: FormData, key: string) {
  const value = String(formData.get(key) ?? "").trim();
  return value === "" ? null : Number(value);
}

function checked(formData: FormData, key: string) {
  return formData.get(key) === "on";
}

function serviceInput(formData: FormData) {
  const recurring = checked(formData, "recurring");
  const deadlineAmount = optionalNumber(formData, "deadline_amount");

  return {
    name: String(formData.get("name") ?? "").trim(),
    folder_id: optionalText(formData, "folder_id"),
    description: optionalText(formData, "description"),
    price: requiredNumber(formData, "price"),
    currency: String(formData.get("currency") ?? "USD").trim().toUpperCase(),
    image_url: null,
    is_active: formData.get("is_active") !== "false",
    is_public: checked(formData, "is_public"),
    recurring,
    recurring_price: recurring ? optionalNumber(formData, "recurring_price") : null,
    recurring_interval_length: recurring
      ? optionalNumber(formData, "recurring_interval_length")
      : null,
    recurring_interval_unit: recurring
      ? optionalText(formData, "recurring_interval_unit")
      : null,
    recurring_action: recurring ? optionalText(formData, "recurring_action") : null,
    group_quantities: checked(formData, "group_quantities"),
    assigned_to_profile_id: optionalText(formData, "assigned_to_profile_id"),
    deadline_amount: deadlineAmount,
    deadline_unit: deadlineAmount == null ? null : optionalText(formData, "deadline_unit"),
    require_client_profile: checked(formData, "require_client_profile"),
    sort_order: requiredNumber(formData, "sort_order"),
  };
}

function folderInput(formData: FormData) {
  return {
    name: String(formData.get("name") ?? "").trim(),
    description: optionalText(formData, "description"),
    parent_id: optionalText(formData, "parent_id"),
    sort_order: requiredNumber(formData, "sort_order"),
  };
}

function formError(error: unknown): CatalogFormState {
  const safe = toClientSafeError(error);
  return {
    error: safe.code === "VALIDATION_ERROR"
      ? "Check the highlighted service settings and try again."
      : safe.message,
  };
}

export async function createServiceAction(
  _state: CatalogFormState,
  formData: FormData,
): Promise<CatalogFormState> {
  void _state;
  const context = await getInternalContext();
  let serviceId: string;

  try {
    const service = await context.services.catalog.create({
      request_id: String(formData.get("request_id") ?? ""),
      ...serviceInput(formData),
    }, context.actor);
    serviceId = service.id;
  } catch (error) {
    return formError(error);
  }

  revalidatePath("/services");
  redirect(`/services/${serviceId}?created=1`);
}

import { randomUUID } from "node:crypto";
import type {

  displayName: string;

export class ServiceRepository {
  constructor(private readonly client: UserMutationClient) {}

  async create(input: CreateServiceInput): Promise<ServiceRow> {
    const { request_id: requestId, ...payload } = input;
    return unwrapOne(
      await this.client.rpc("app_create_service", {
        p_request_id: requestId ?? randomUUID(),
        p_payload: payload,
      }),
      "Service",
    );
  }

  async duplicate(id: string, requestId: string = randomUUID()): Promise<ServiceRow> {
    return unwrapOne(
      await this.client.rpc("app_duplicate_service", {
        p_request_id: requestId,
        p_service_id: id,
      }),
      "Service",
    );
  }

  async getById(id: string): Promise<ServiceRow> {
    return unwrapOne(
      await this.client.from("services").select().eq("id", id).is("archived_at", null).single(),
      "Service",
    );
  }

  async listActive(): Promise<ServiceRow[]> {
    return unwrapMany(
      await this.client
        .from("services")
        .select()
        .eq("is_active", true)
        .is("archived_at", null)
        .order("name"),
    );
  }

  async listCatalog(): Promise<ServiceRow[]> {
    return unwrapMany(
      await this.client
        .from("services")
        .select()
        .is("archived_at", null)
        .order("sort_order")
        .order("name"),
    );
  }

  /**
   * The client Store slice of the catalog. RLS already narrows a client session
   * to public, active, non-archived services; the predicates are restated here
   * so the query means the same thing whoever runs it.
   */
  async listStoreServices(): Promise<ServiceRow[]> {
    return unwrapMany(
      await this.client
        .from("services")
        .select()
        .eq("is_public", true)
        .eq("is_active", true)
        .is("archived_at", null)
        .order("sort_order")
        .order("name"),
    );
  }

  async getStoreService(id: string): Promise<ServiceRow> {
    return unwrapOne(
      await this.client
        .from("services")
        .select()
        .eq("id", id)
        .eq("is_public", true)
        .eq("is_active", true)
        .is("archived_at", null)
        .single(),
      "Service",
    );
  }

  async update(id: string, input: UpdateServiceInput): Promise<ServiceRow> {
    return unwrapOne(
      await this.client.rpc("app_update_service", { p_service_id: id, p_patch: input }),
      "Service",
    );
  }

  async archive(id: string): Promise<ServiceRow> {
    return unwrapOne(
      await this.client.rpc("app_archive_service", { p_service_id: id }),
      "Service",
    );
  }

  async createFolder(input: CreateServiceFolderInput): Promise<ServiceFolderRow> {
    return unwrapOne(
      await this.client.rpc("app_create_service_folder", {
        p_request_id: input.request_id ?? randomUUID(),
        p_name: input.name,
        p_description: input.description ?? undefined,
        p_parent_id: input.parent_id ?? undefined,
        p_sort_order: input.sort_order,
      }),
      "Service folder",
    );
  }

  async getFolderById(id: string): Promise<ServiceFolderRow> {
    return unwrapOne(
      await this.client
        .from("service_folders")
        .select()
        .eq("id", id)
        .is("archived_at", null)
        .single(),
      "Service folder",
    );
  }

  async listFolders(): Promise<ServiceFolderRow[]> {
    return unwrapMany(
      await this.client
        .from("service_folders")
        .select()
        .is("archived_at", null)
        .order("sort_order")
        .order("name"),
    );
  }

  async updateFolder(id: string, input: UpdateServiceFolderInput): Promise<ServiceFolderRow> {
    return unwrapOne(
      await this.client.rpc("app_update_service_folder", {
        p_folder_id: id,
        p_patch: input,
      }),
      "Service folder",
    );
  }

  async archiveFolder(id: string): Promise<ServiceFolderRow> {
    return unwrapOne(
      await this.client.rpc("app_archive_service_folder", { p_folder_id: id }),
      "Service folder",
    );
  }

  async listAssignees(): Promise<ServiceAssigneeOption[]> {
    const [profiles, roles] = await Promise.all([
      this.client
        .from("profiles")
        .select("id, display_name, role_id")
        .is("archived_at", null),
      this.client.from("roles").select("id, name"),
    ]);
    const internalRoleIds = new Set(
      unwrapMany(roles)
        .filter((role) => ["owner", "admin", "manager", "staff"].includes(role.name))
        .map((role) => role.id),
    );

    return unwrapMany(profiles)
      .filter((profile) => internalRoleIds.has(profile.role_id))
      .map((profile) => ({
        id: profile.id,
        displayName: profile.display_name ?? "Unnamed teammate",
      }))
      .sort((left, right) => left.displayName.localeCompare(right.displayName));
  }

const positiveIntegerSchema = z.number().int().positive();

const currencySchema = z.string().regex(/^[A-Z]{3}$/, "Use a three-letter currency code.");

  description: nullableNonBlankTextSchema.optional(),

  recurring_action: z.enum(recurringActions).nullable().optional(),
  group_quantities: z.boolean().optional(),
  assigned_to_profile_id: uuidSchema.nullable().optional(),
  deadline_amount: positiveIntegerSchema.nullable().optional(),
  deadline_unit: z.enum(deadlineUnits).nullable().optional(),
  require_client_profile: z.boolean().optional(),
  sort_order: sortOrderSchema.optional(),
});

function validateServiceConfiguration(
  value: {
    recurring?: boolean;
    recurring_price?: number | null;
    recurring_interval_length?: number | null;
    recurring_interval_unit?: (typeof recurringIntervalUnits)[number] | null;
    recurring_action?: (typeof recurringActions)[number] | null;
    deadline_amount?: number | null;
    deadline_unit?: (typeof deadlineUnits)[number] | null;
  },
  context: z.RefinementCtx,
) {
  const recurrenceValues = [
    value.recurring_price,
    value.recurring_interval_length,
    value.recurring_interval_unit,
    value.recurring_action,
  ];

  if (value.recurring && recurrenceValues.some((candidate) => candidate == null)) {
    context.addIssue({
      code: "custom",
      path: ["recurring"],
      message: "Recurring services require a price, interval, and action.",
    });
  }

  if (value.recurring === false && recurrenceValues.some((candidate) => candidate != null)) {
    context.addIssue({
      code: "custom",
      path: ["recurring"],
      message: "Disable recurring settings when the service is not recurring.",
    });
  }

  if ((value.deadline_amount == null) !== (value.deadline_unit == null)) {
    context.addIssue({
      code: "custom",
      path: ["deadline_amount"],
      message: "Deadline amount and unit must be supplied together.",
    });
  }
}

    recurring: z.boolean().default(false),
    group_quantities: z.boolean().default(false),
    require_client_profile: z.boolean().default(false),
    sort_order: sortOrderSchema.default(0),
  })
  .superRefine(validateServiceConfiguration);

export const updateServiceSchema = serviceFieldsSchema
  .partial()
  .superRefine((value, context) => {
    if (Object.keys(value).length === 0) {
      context.addIssue({
        code: "custom",
        message: "At least one service field must be supplied.",
      });
      return;
    }
    validateServiceConfiguration(value, context);
  });
